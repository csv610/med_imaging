\PageBanner{Doppler Ultrasound (Vascular \& Hemodynamic)}{1. BEFORE THE SCAN}{Form 1 of 4}

\begin{GuidanceBox}
{\fontsize{8.1}{10.1}\selectfont
\textbf{\color{Navy}What to check first:} Confirm vascular territory (venous, arterial, or organ perfusion), side, pretest probability (e.g., Wells score/D-dimer for DVT, ABI for PAD), and prior vascular imaging.\par\vspace{1.5pt}
\textbf{\color{Navy}When to proceed or defer:} Proceed when Doppler findings will change management. In low-probability DVT with negative D-dimer, deferral is usually appropriate.\par\vspace{1.5pt}
\textbf{\color{Navy}Adaptations:} Adapt positioning for frailty, pain, casts, or wounds; note anticoagulation, arrhythmias, or grafts that affect flow interpretation.}
\end{GuidanceBox}

\PageBanner{Doppler Ultrasound (Vascular \& Hemodynamic)}{3. WHAT THE IMAGE SHOWS \& MEANS}{Form 3 of 4}

\renewcommand{\arraystretch}{1.2}
\noindent\begin{tabularx}{\linewidth}{@{}>{\bfseries\color{Navy}}p{3.0cm} >{\color{Navy}}p{5.4cm} X@{}}
\toprule
\textbf{Context} & \textbf{Key Doppler Finding} & \textbf{Typical Meaning / Interpretation} \\
\midrule
Venous (DVT) & Noncompressible vein; absent color/phasicity & Supports deep vein thrombosis or proximal block \\
Arterial Stenosis & Focal PSV jump, aliasing, elevated PSV ratio & Supports haemodynamically significant stenosis \\
Distal Waveform & Tardus-parvus (blunted upstroke, slow rise) & Suggests upstream stenosis or occlusion \\
Organ / Torsion & High RI, reversed/absent diastolic or no flow & High vascular resistance, rejection, or torsion \\
Venous Reflux & Retrograde flow ($>0.5$ s superf.; $>1.0$ s deep) & Supports venous valvular insufficiency \\
\bottomrule
\end{tabularx}\par\vspace{4pt}

\begin{GuidanceBox}
{\fontsize{8.1}{10.1}\selectfont
\textbf{\color{Navy}What Doppler leaves unanswered:} Angle errors ($>60^\circ$), heavy arterial calcification (shadowing), vessel depth, low-flow states, or arrhythmias can distort velocities. A negative proximal DVT scan may require repeat imaging under the pathway; CTA/MRA may be needed for intervention planning.\par\vspace{2pt}
\textbf{\color{Navy}Worked interpretation example:} Finding: Vein fails to compress with absent color fill. Interpretation: supports thrombosis. If absent, verify coverage and pretest probability.\par\vspace{2pt}
\textbf{\color{Navy}Illustration guide (4:3 plate):} Pair B-mode/color view with the spectral Doppler trace; label vessel, sample gate, angle ($\le 60^\circ$), PSV, and EDV.}
\end{GuidanceBox}
