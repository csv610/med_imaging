\PageBanner{General Ultrasound (Sonography \& Doppler)}{1. BEFORE THE SCAN}{Form 1 of 4}

\begin{GuidanceBox}
{\fontsize{8.1}{10.1}\selectfont
\textbf{\color{Navy}What to check first:} Confirm anatomical region, side, clinical question, and required preparation (fasting, full bladder) or prior labs (e.g., D-dimer, LFTs, $\beta$-hCG).\par\vspace{1.5pt}
\textbf{\color{Navy}When to proceed or defer:} Proceed when ultrasound is the appropriate first-line test (no ionising radiation) and will change care. Defer if a low-risk pathway (e.g., negative D-dimer in unlikely DVT) avoids imaging.\par\vspace{1.5pt}
\textbf{\color{Navy}Adaptations:} Adapt positioning for frailty, pain, dressings, or wounds; obtain consent for endocavitary views; apply ALARA/thermal index limits in pregnancy and pediatrics.}
\end{GuidanceBox}

\midrule
Vascular / DVT & Noncompressible vein; intraluminal material & Supports deep vein thrombosis \\

Renal / Urinary & Dilated collecting system; absent ureteric jet & Supports urinary tract obstruction \\
Cyst vs. Solid & Anechoic, thin-walled, posterior enhancement & Simple cyst/fluid vs. complex or solid lesion \\
Doppler / Flow & Absent, reversed, or tardus-parvus waveform & Supports occlusion, torsion, or stenosis \\
\bottomrule
\end{tabularx}\par\vspace{4pt}

\begin{GuidanceBox}
{\fontsize{8.1}{10.1}\selectfont
\textbf{\color{Navy}What ultrasound leaves unanswered:} Acoustic shadowing from bowel gas or bone, body habitus, and operator/window dependence can limit visualisation. A negative proximal DVT scan may require repeat imaging under the pathway; equivocal biliary or deep abdominal findings may need HIDA, CT, or MRI.\par\vspace{2pt}
\textbf{\color{Navy}Worked interpretation example:} Finding: Vein fails to compress. Interpretation: key sign supporting thrombosis. If absent, check scan coverage and pretest probability before closing the question.\par\vspace{2pt}
\textbf{\color{Navy}Illustration guide (4:3 plate):} Show paired views (e.g., without vs. with compression, or grayscale vs. color Doppler) with the target structure and measurements labelled.}
\end{GuidanceBox}
